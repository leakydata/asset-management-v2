================================================================================
frmCatQuickLook — build it once, then paste the code below
================================================================================

A UserForm is a .frm plus a binary .frx companion, so it can't be handed over
as importable text the way a .bas can. Building it by hand takes a couple of
minutes and only happens once.

--------------------------------------------------------------------------------
1. CREATE THE FORM
--------------------------------------------------------------------------------

In the VBE:  Insert ▸ UserForm

In the Properties window (F4) set ONE property:

    (Name)      frmCatQuickLook

Nothing else. Size, caption and button text are all set by the code — a fresh
UserForm defaults to about 240x180, which looks tiny, and the code fixes that
on open.

--------------------------------------------------------------------------------
2. DROP ON SIX CONTROLS
--------------------------------------------------------------------------------

Names must match exactly — the code is wired to them. **Position and size do
not matter at all** — drop them anywhere, the code lays them out on open. You
don't need to set captions either.

    Control          (Name)
    ------------------------------------
    TextBox          txtSerial
    CommandButton    btnLookup
    ListBox          lstRecords
    ListBox          lstFields
    CommandButton    btnCopy
    CommandButton    btnPaste

OPTIONAL, if you want headings between the two lists — add them or don't, the
code lays out correctly either way and captions them for you:

    Label            lblRecords      -> "Ownership records"
    Label            lblFields       -> "Record detail"

DO NOT put the controls inside a Frame. A control in a Frame has its Left/Top
measured from the FRAME, not the form, so the absolute positioning the code
does lands it in the wrong place and everything piles up. The code checks for
this on open and tells you which controls are affected. To move one out: cut
it (Ctrl+X), click the form background, paste (Ctrl+V).

Rough layout:

    ┌─ Cat Quick Look ─────────────────────────────┐
    │ [txtSerial................]  [ btnLookup ]   │
    │ ┌─ lstRecords ─────────────────────────────┐ │
    │ │ C01913004  DUSHORE SEWER  [OWNED/ACTIVE] │ │
    │ │ C10001501  DUSHORE BOROUGH [OWNED/ACTIVE]│ │
    │ └──────────────────────────────────────────┘ │
    │ ┌─ lstFields ──────────────────────────────┐ │
    │ │ SerialNumber      KR416089               │ │
    │ │ MakeCode          CAT                    │ │
    │ │ DCN               C01913004              │ │
    │ │ OwnershipType     OWNED                  │ │
    │ │ ...                                      │ │
    │ └──────────────────────────────────────────┘ │
    │ [ btnCopy ]   [ btnPaste                   ] │
    └──────────────────────────────────────────────┘

--------------------------------------------------------------------------------
3. PASTE THE CODE
--------------------------------------------------------------------------------

Right-click the form ▸ View Code, then paste everything below the line.
(Do NOT put this in a normal module — it belongs to the form.)

================================================================================
Option Explicit

' Set to False once the layout is right. Module-level declarations have to sit
' up here in the declarations section - VBA will not see a Const declared
' between two procedures.
Private Const DIAG As Boolean = True

' One entry per ownership record; each item is the array RecordValues returns.
Private mRows As Collection

'==============================================================================
' Setup
'==============================================================================
Private Sub UserForm_Initialize()
    Dim stray As String
    stray = ControlsNotOnForm()
    If Len(stray) > 0 Then
        ' Worth stopping for: a control inside a Frame has coordinates relative
        ' to the FRAME, so the absolute positioning below silently lands it in
        ' the wrong place and everything looks piled up.
        MsgBox "These controls are inside a Frame or other container:" & vbCrLf & vbCrLf & _
               "    " & stray & vbCrLf & vbCrLf & _
               "Their Left/Top are measured from the container, not the form, so " & _
               "the layout will be wrong. Cut them (Ctrl+X), click the form " & _
               "background, and paste (Ctrl+V) to move them out - then delete " & _
               "the Frame." & vbCrLf & vbCrLf & _
               "For headings above the lists, add Labels named lblRecords and " & _
               "lblFields instead; the code positions them if they exist.", _
               vbExclamation, "Cat Quick Look - form layout"
    End If

    LayoutControls

    lstRecords.ColumnCount = 6
    lstFields.ColumnCount = 2

    ' Enter in the serial box triggers Look Up.
    btnLookup.Default = True

    ResetDisplay
End Sub

' Names any of the six that were dropped into a container rather than straight
' onto the form.
Private Function ControlsNotOnForm() As String
    Dim nm As Variant, c As Object
    For Each nm In Array("txtSerial", "btnLookup", "lstRecords", _
                         "lstFields", "btnCopy", "btnPaste")
        Set c = Nothing
        On Error Resume Next
        Set c = Me.Controls(CStr(nm))
        On Error GoTo 0
        If Not c Is Nothing Then
            If Not c.Parent Is Me Then
                ControlsNotOnForm = ControlsNotOnForm & _
                    IIf(Len(ControlsNotOnForm) > 0, ", ", "") & CStr(nm)
            End If
        End If
    Next nm
End Function

' Everything about the form's appearance is set here rather than in the
' designer, so building it by hand is only ever "drop the controls and name
' them". Wherever you dragged them, they end up here. A new UserForm defaults
' to roughly 240x180, which is why an unsized one looks tiny.
'
' Do NOT put the controls in a Frame - see ControlsNotOnForm above. If you want
' headings between the lists, add two Labels named lblRecords and lblFields and
' they get positioned and captioned automatically. Leave them out and the form
' closes the gap.
Private Sub LayoutControls()
    Const PAD As Long = 8
    Const BTN As Long = 26
    Const LBL As Long = 13
    ' Every full-width control spans PAD .. PAD + W, so the serial row, both
    ' lists and the button row share one right edge.
    Const W   As Long = 460          ' usable width inside the border

    ' MSForms has no "fit the text to the box" - AutoSize resizes the BOX to
    ' the text, which is the wrong way round. You set Font.Size and make the
    ' control tall enough for it. A control needs roughly Font.Size + 10 points
    ' of height, which is why a default TextBox is 18 high: Tahoma 8pt + 10.
    Const UI_SIZE   As Single = 11   ' the serial box
    Const LIST_SIZE As Single = 9    ' the two lists
    Const REC_H     As Long = 72     ' ~6 records before it scrolls
    Const FLD_H     As Long = 300    ' all 22 fields, no scrolling
    Dim ROW As Long: ROW = UI_SIZE + 11

    ' SIZE THE FORM FIRST, and MEASURE its chrome rather than guessing it.
    ' Me.Width/Height are the OUTER window; the usable area is InsideWidth /
    ' InsideHeight, and the difference (border + title bar) varies with the
    ' Windows theme and DPI. Guessing it is what left a band of dead grey down
    ' the right and along the bottom: the controls were laid out to a width the
    ' form did not actually have inside.
    Dim labels As Long
    labels = LabelSpace("lblRecords", LBL) + LabelSpace("lblFields", LBL)

    Dim contentH As Long
    contentH = PAD + ROW + PAD + labels + REC_H + PAD + FLD_H + PAD + BTN + PAD

    ' Provisional size, so InsideWidth/InsideHeight are meaningful to read.
    Me.Width = W + (PAD * 2) + 20
    Me.Height = contentH + 40

    Dim chromeW As Long: chromeW = Me.Width - Me.InsideWidth
    Dim chromeH As Long: chromeH = Me.Height - Me.InsideHeight

    Me.Width = W + (PAD * 2) + chromeW
    Me.Height = contentH + chromeH

    Me.Caption = "Cat Quick Look"

    btnLookup.Caption = "Look Up"
    btnCopy.Caption = "Copy Row"
    btnPaste.Caption = "Paste at Selection"

    ' Monospaced for anything holding an identifier. Serials and DCNs get read
    ' character by character and 0/O, 1/l are easy to confuse in a proportional
    ' face; a fixed-width one also lines the value column up.
    txtSerial.Font.Name = "Consolas"
    txtSerial.Font.Size = UI_SIZE
    lstRecords.Font.Name = "Consolas"
    lstRecords.Font.Size = LIST_SIZE
    lstFields.Font.Name = "Consolas"
    lstFields.Font.Size = LIST_SIZE

    ' Lay out against the measured client area, not the constant. If anything
    ' clamps or rounds, every full-width control still shares one right edge.
    Dim FW As Long: FW = Me.InsideWidth - (PAD * 2)

    ' A running cursor down the form, so an optional label simply consumes a
    ' band of space and everything below shuffles down.
    Dim y As Long: y = PAD

    Dim lookLeft As Long: lookLeft = PAD + (FW - 100) + PAD

    Place txtSerial, PAD, y, FW - 100, ROW
    Place btnLookup, lookLeft, y, (PAD + FW) - lookLeft, ROW
    y = y + ROW + PAD

    ' MSForms only draws real column heads for a RowSource-bound list, never
    ' for one filled with AddItem - so the heading is a Label in the same
    ' monospaced face, captioned after the widths are known (below).
    y = y + LabelSpace("lblRecords", LBL)
    Place lstRecords, PAD, y, FW, REC_H
    y = y + REC_H + PAD

    y = y + PlaceLabel("lblFields", "Record detail", PAD, y, FW, LBL)
    Place lstFields, PAD, y, FW, FLD_H
    y = y + FLD_H + PAD

    Place btnCopy, PAD, y, 110, BTN
    Place btnPaste, PAD + 110 + PAD, y, FW - 110 - PAD, BTN

    ' Match the lists to the right edge the button row actually achieved,
    ' rather than trusting that the arithmetic above landed.
    Dim rightEdge As Long
    rightEdge = btnPaste.Left + btnPaste.Width
    If btnLookup.Left + btnLookup.Width < rightEdge Then
        rightEdge = btnLookup.Left + btnLookup.Width
    End If
    Place lstRecords, PAD, lstRecords.Top, rightEdge - PAD, REC_H
    Place lstFields, PAD, lstFields.Top, rightEdge - PAD, FLD_H

    ' Columns are a fraction of what the list really is, not of the constant.
    lstRecords.ColumnWidths = RecordColumnWidths(lstRecords.Width)
    lstFields.ColumnWidths = CStr(Int(lstFields.Width * 0.32)) & " pt;" & _
                             CStr(Int(lstFields.Width * 0.55)) & " pt"

    PlaceLabel "lblRecords", RecordHeader(LIST_SIZE, lstRecords.Width), _
               PAD, lstRecords.Top - (LBL + 2), rightEdge - PAD, LBL, _
               "Consolas", LIST_SIZE

    If DIAG Then ReportLayout FW, rightEdge
End Sub

' Immediate window (Ctrl+G in the VBE), not a MsgBox - you can select and
' copy from it.
Private Sub ReportLayout(ByVal FW As Long, ByVal rightEdge As Long)
    Debug.Print String$(50, "-")
    Debug.Print "form   Width=" & Me.Width & "  InsideWidth=" & Me.InsideWidth
    Debug.Print "       Height=" & Me.Height & "  InsideHeight=" & Me.InsideHeight
    Debug.Print "FW=" & FW & "   rightEdge asked for=" & rightEdge
    Debug.Print
    Debug.Print "control        Left  Width  right   Top  Height"
    Debug.Print String$(50, "-")
    Debug.Print Row("txtSerial", txtSerial);
    Debug.Print Row("btnLookup", btnLookup);
    Debug.Print Row("lstRecords", lstRecords);
    Debug.Print Row("lstFields", lstFields);
    Debug.Print Row("btnCopy", btnCopy);
    Debug.Print Row("btnPaste", btnPaste);
End Sub

Private Function Row(ByVal nm As String, ByVal c As Object) As String
    Row = Left$(nm & Space$(15), 15) & _
          Right$(Space$(4) & CStr(CLng(c.Left)), 5) & _
          Right$(Space$(6) & CStr(CLng(c.Width)), 7) & _
          Right$(Space$(6) & CStr(CLng(c.Left + c.Width)), 7) & _
          Right$(Space$(5) & CStr(CLng(c.Top)), 6) & _
          Right$(Space$(6) & CStr(CLng(c.Height)), 8) & vbCrLf
End Function

' Vertical space an optional label will take, or 0 if it isn't on the form.
' Needed up front so the form can be sized before anything is positioned.
Private Function LabelSpace(ByVal nm As String, ByVal h As Long) As Long
    Dim c As Object
    On Error Resume Next
    Set c = Me.Controls(nm)
    On Error GoTo 0
    If Not c Is Nothing Then LabelSpace = h + 2
End Function

' Positions an OPTIONAL label and returns the vertical space it used, or 0 if
' you never added it. The form lays out correctly either way.
Private Function PlaceLabel(ByVal nm As String, ByVal cap As String, _
                            ByVal x As Long, ByVal y As Long, _
                            ByVal w As Long, ByVal h As Long, _
                            Optional ByVal fontName As String = "", _
                            Optional ByVal fontSize As Single = 0) As Long
    Dim c As Object
    On Error Resume Next
    Set c = Me.Controls(nm)
    On Error GoTo 0
    If c Is Nothing Then Exit Function

    If Len(fontName) > 0 Then c.Font.Name = fontName
    If fontSize > 0 Then c.Font.Size = fontSize
    c.Caption = cap
    Place c, x, y, w, h
    PlaceLabel = h + 2
End Function

' Positions AND sizes a control in one call.
'
' Assigning .Width and .Height individually was silently doing nothing on the
' two ListBoxes - no error, the value simply did not stick, while .Left and
' .Top on the same control worked. .Move sets all four through a single call
' and is the documented way to reposition an MSForms control.
'
' AutoSize is tried first because a TextBox, Label or CommandButton with it on
' resizes to its contents and ignores an explicit size. ListBox has no such
' property, hence the guard rather than a plain assignment.
Private Sub Place(ByVal c As Object, ByVal x As Long, ByVal y As Long, _
                  ByVal w As Long, ByVal h As Long)
    On Error Resume Next
    c.AutoSize = False
    On Error GoTo 0

    c.Move x, y, w, h
End Sub

' Called by CatQuickLook before Show, so selecting a cell that holds a serial
' and clicking the button pre-fills it.
Public Sub SeedSerial(ByVal s As String)
    txtSerial.Text = s
End Sub

Private Sub UserForm_Activate()
    txtSerial.SetFocus
    If Len(txtSerial.Text) > 0 Then txtSerial.SelStart = 0: txtSerial.SelLength = Len(txtSerial.Text)
End Sub

'==============================================================================
' Look up
'==============================================================================
Private Sub btnLookup_Click()
    Dim serial As String
    serial = CleanId(txtSerial.Text)
    If Len(serial) = 0 Then
        Me.Caption = "Cat Quick Look - enter a serial number"
        Exit Sub
    End If

    ResetDisplay
    Me.Caption = "Cat Quick Look - looking up " & serial & " ..."
    DoEvents

    Dim txt As String
    On Error GoTo Fail
    Application.Cursor = xlWait
    txt = CatSearch(serial, "")          ' raises on a non-200
    Application.Cursor = xlDefault
    On Error GoTo 0

    Dim recs As Object
    Set recs = OwnershipRecords(txt)
    If recs Is Nothing Then
        Me.Caption = "Cat Quick Look - unexpected response"
        Exit Sub
    End If
    If recs.Count = 0 Then
        ' The call succeeded and returned nothing - that is a real answer, not
        ' a failure, and worth saying differently.
        Me.Caption = "Cat Quick Look - " & serial & " is NOT in CCAT"
        Exit Sub
    End If

    Set mRows = New Collection
    Dim i As Long, v As Variant
    For i = 1 To recs.Count
        v = RecordValues(recs(i))
        mRows.Add v
        ' One column each, so nothing shifts when a DCN name is long
        lstRecords.AddItem CStr(v(2))                    ' DCN
        lstRecords.List(lstRecords.ListCount - 1, 1) = CStr(v(11))          ' DcnName
        lstRecords.List(lstRecords.ListCount - 1, 2) = CStr(v(3))           ' OwnershipType
        lstRecords.List(lstRecords.ListCount - 1, 3) = CStr(v(6))           ' Status
        lstRecords.List(lstRecords.ListCount - 1, 4) = YesNo(CStr(v(10)))   ' HasSubscription
        lstRecords.List(lstRecords.ListCount - 1, 5) = CStr(v(14))          ' DealerCode
    Next i

    Me.Caption = "Cat Quick Look - " & recs.Count & " record(s) for " & serial
    lstRecords.ListIndex = 0             ' fires lstRecords_Click -> ShowRecord
    Exit Sub

Fail:
    Application.Cursor = xlDefault
    Me.Caption = "Cat Quick Look - FAILED: " & Err.Description
End Sub

Private Sub lstRecords_Click()
    ShowRecord
End Sub

Private Sub ShowRecord()
    lstFields.Clear
    If mRows Is Nothing Then Exit Sub
    If lstRecords.ListIndex < 0 Then Exit Sub

    Dim v As Variant: v = mRows(lstRecords.ListIndex + 1)
    Dim h As Variant: h = HeaderArray()

    Dim i As Long
    For i = LBound(h) To UBound(h)
        lstFields.AddItem CStr(h(i))
        lstFields.List(lstFields.ListCount - 1, 1) = CStr(v(i))
    Next i
End Sub

Private Sub ResetDisplay()
    Set mRows = Nothing
    lstRecords.Clear
    lstFields.Clear
End Sub

'==============================================================================
' Copy / paste
'==============================================================================

' Whole record as two tab-separated lines (headers, then values) - drops into
' Excel, an email or a ticket unchanged.
Private Sub btnCopy_Click()
    Dim v As Variant
    If Not PickedRecord(v) Then Exit Sub

    Dim h As Variant: h = HeaderArray()
    Dim s As String, i As Long
    For i = LBound(h) To UBound(h)
        s = s & CStr(h(i)) & IIf(i < UBound(h), vbTab, "")
    Next i
    s = s & vbCrLf
    For i = LBound(v) To UBound(v)
        s = s & CStr(v(i)) & IIf(i < UBound(v), vbTab, "")
    Next i

    ' MSForms.DataObject is available because this project has a UserForm.
    Dim d As Object
    Set d = New MSForms.DataObject
    d.SetText s
    d.PutInClipboard

    Me.Caption = "Cat Quick Look - copied all " & (UBound(h) + 1) & " columns"
End Sub

' Writes into whatever cell is selected RIGHT NOW - resolved at click time, so
' it is never stale no matter how much you clicked around while this was open.
Private Sub btnPaste_Click()
    Dim v As Variant
    If Not PickedRecord(v) Then Exit Sub

    Dim c As Range
    On Error Resume Next
    Set c = Application.ActiveCell
    On Error GoTo 0
    If c Is Nothing Then
        Me.Caption = "Cat Quick Look - select a cell in a workbook first"
        Exit Sub
    End If

    Dim n As Long
    n = QuickLookPasteWidth(c.Worksheet)
    QuickLookPasteAt c, v, n

    Me.Caption = "Cat Quick Look - pasted " & n & " cols to " & _
                 c.Worksheet.Name & "!" & c.Address(0, 0)
End Sub

' The record list's column widths, in points. ONE definition - the list and
' its heading are both built from this, so they cannot drift apart.
'
' The total is deliberately ~30pt short of the 400pt list: a ListBox grows its
' own vertical scrollbar once items overflow, and that steals about 17pt.
' Without the slack the last column would clip exactly when a serial has
' enough records to need scrolling.
Private Function RecordColumns() As Variant
    '            DCN  name type status sub dealer
    RecordColumns = Array(68, 155, 58, 52, 30, 38)
End Function

' Scaled to the list's real width, keeping ~13% clear for the vertical
' scrollbar so the last column never clips.
Private Function RecordColumnWidths(ByVal listW As Long) As String
    Dim w As Variant: w = RecordColumns()
    Dim k As Single: k = (listW * 0.87) / SumOf(w)
    Dim s As String, i As Long
    For i = LBound(w) To UBound(w)
        s = s & CStr(Int(w(i) * k)) & " pt" & IIf(i < UBound(w), ";", "")
    Next i
    RecordColumnWidths = s
End Function

Private Function SumOf(ByVal w As Variant) As Long
    Dim i As Long
    For i = LBound(w) To UBound(w): SumOf = SumOf + w(i): Next i
End Function

' A heading padded to sit over the columns. Consolas advances 0.55 em per
' character, so a column w points wide holds Int(w / (size * 0.55)) of them.
Private Function RecordHeader(ByVal fontSize As Single, ByVal listW As Long) As String
    Dim w As Variant: w = RecordColumns()
    Dim caps As Variant
    caps = Array("DCN", "DCN NAME", "TYPE", "STATUS", "SUB", "DLR")

    Dim k As Single: k = (listW * 0.87) / SumOf(w)
    Dim charW As Single: charW = fontSize * 0.55
    Dim s As String, i As Long, n As Long
    For i = LBound(w) To UBound(w)
        n = Int((w(i) * k) / charW)
        s = s & Left$(CStr(caps(i)) & Space$(n), n)
    Next i
    RecordHeader = s
End Function

' TRUE/FALSE -> Yes/No, and a dash when the API didn't return the field at
' all. Absent is not the same as No, so it must not read as one.
Private Function YesNo(ByVal s As String) As String
    Select Case UCase$(Trim$(s))
        Case "TRUE":  YesNo = "Yes"
        Case "FALSE": YesNo = "No"
        Case Else:    YesNo = "-"
    End Select
End Function

' Hands back the selected record, or reports why it can't.
' NOT called Selected: UserForm already inherits a member of that name,
' and redeclaring it is "Member already exists in an object module from
' which this object module derives".
Private Function PickedRecord(ByRef v As Variant) As Boolean
    If mRows Is Nothing Then
        Me.Caption = "Cat Quick Look - look up a serial first"
        Exit Function
    End If
    If lstRecords.ListIndex < 0 Then
        Me.Caption = "Cat Quick Look - pick a record from the list first"
        Exit Function
    End If
    v = mRows(lstRecords.ListIndex + 1)
    PickedRecord = True
End Function
